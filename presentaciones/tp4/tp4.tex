% Modelo general: segunda ley de Newton con fuerzas de a pares y la ley de contacto (resorte
% normal; pedido 2026-10-19: va acá y no en Simulaciones) + avance en un paso + el esquema del
% billar, Velocity Verlet. Los otros tres esquemas del oscilador no van (pedido 2026-10-19).
% Obstáculos y pared (imagen) son del sistema particular: van en "Sistema: interacción".
\begin{frame}{Modelo: ecuación de movimiento y esquema de integración}
  \setlength{\abovedisplayskip}{3pt}\setlength{\belowdisplayskip}{3pt}
  \textbf{Segunda ley de Newton con contactos elásticos}
  \[
    m_i\,\vect{a}_i(t) = \vect{F}_i(t) = \sum_{j\,:\,\xi_{ij} > 0} -k\,\xi_{ij}\,\hat{\vect{e}}_{ij},
    \qquad
    \xi_{ij} = 2r - |\vect{r}_j - \vect{r}_i|, \quad
    \hat{\vect{e}}_{ij} = \frac{\vect{r}_j - \vect{r}_i}{|\vect{r}_j - \vect{r}_i|}
  \]
  \vspace{0.3em}
  \textbf{Avance en un paso $dt$ con aceleración constante}
  \[
    \vect{r}_i(t + dt) = \vect{r}_i(t) + \vect{v}_i(t)\,dt + \tfrac{1}{2}\,\vect{a}_i(t)\,dt^2,
    \qquad
    \vect{v}_i(t + dt) = \vect{v}_i(t) + \vect{a}_i(t)\,dt
  \]
  \vspace{0.3em}
  \textbf{Velocity Verlet}
  \begin{align*}
    \vect{r}_i(t + dt) &= \vect{r}_i(t) + \vect{v}_i(t)\,dt + \tfrac{1}{2}\,\vect{a}_i(t)\,dt^2\\
    \vect{v}_i(t + dt) &= \vect{v}_i(t) + \tfrac{1}{2}\bigl[\vect{a}_i(t) + \vect{a}_i(t + dt)\bigr]\,dt
  \end{align*}
\end{frame}

% histograma con las partículas de las 10 realizaciones (y de todos los estados t >= t_est en la
% estacionaria) y un único promedio de la rapidez; mismos valores que promediar por realización
% (igual N en todas), sin barras de error.
\begin{frame}{Observables: distribución de velocidades}
  \begin{itemize}\setlength{\itemsep}{0.6em}
    \item $N_b(t)$: partículas de las 10 realizaciones con rapidez en el bin $b$ (centro $v_b$,
      ancho $\Delta v$):
      \[
        f(v_b, t) = \frac{N_b(t)}{10\,N\,\Delta v}, \qquad
        \bar{v}(t) = \frac{1}{10\,N}\sum_{k=1}^{10}\sum_{i=1}^{N} |\vect{v}_{i,k}(t)|
      \]
    \item $f(v_b)$ estacionaria: promedio de $f(v_b, t)$ en $t \ge t_{\mathrm{est}}$
    \item Ajuste de Maxwell-Boltzmann 2D: $\kT^{*}$ minimiza $\mathcal{E}(\kT)$
      \[
        f_{\mathrm{MB}}(v) = \frac{m\,v}{\kT}\,\exp\!\left(-\frac{m\,v^2}{2\,\kT}\right), \qquad
        \mathcal{E}(\kT) = \sum_b \bigl[f(v_b) - f_{\mathrm{MB}}(v_b)\bigr]^2
      \]
  \end{itemize}
\end{frame}

\begin{frame}{Conclusiones}
  \small
  \begin{itemize}\setlength{\itemsep}{0.5em}
    \item Oscilador: Beeman da el menor ECM, por poco, en todo el rango de $dt$; Beeman,
      Verlet original y Velocity Verlet, de $10^{4}$ a $10^{12}$ veces menor que Euler
      predictor-corrector.
    \item Con $dt = 10^{-5}\uni{s}$ la energía se conserva con
      $\langle\varepsilon\rangle = (5 \pm 2)\ten{-6}$; con $dt \ge 10^{-3}\uni{s}$ diverge.
    \item Costo: eventos es más rápido hasta $N = 400$ y paso temporal desde $N = 500$
      ($100\uni{s}$ contra $850\uni{s}$ en $N = 675$).
    \item Densidad: $\langle t_{90} \rangle$ mínimo, $23 \pm 4\uni{s}$, en
      $\rho = 61\uni{m^{-2}}$; $1270 \pm 70\uni{s}$ en $\rho = 765\uni{m^{-2}}$; desde
      $\rho = 795\uni{m^{-2}}$ no llega a $F_u = 0.9$ en $1500\uni{s}$.
    \item Obstáculos ($N = 100$): $\langle t_{90} \rangle \approx 15\uni{s}$ con $x_o$ entre
      $0.15$ y $0.35\uni{m}$, contra $23 \pm 3\uni{s}$ en $x_o = r$ y $32 \pm 5\uni{s}$ en
      $x_o = R - r$; la misma meseta con $N = 60$ y $20$.
    \item Velocidades: estacionario desde $t = 1\uni{s}$; Maxwell-Boltzmann con
      $\kT^{*} = 12.41\uni{mJ}$, 0.7\,\% debajo de $m v_0^2/2$.
  \end{itemize}
\end{frame}
