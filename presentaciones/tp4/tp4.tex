% Modelo general: segunda ley de Newton con fuerzas de a pares y la ley de contacto (resorte
% normal; pedido 2026-10-19: va acá y no en Simulaciones) + avance en un paso, que es el esquema
% del billar, Velocity Verlet (2026-10-23: se sacó el bloque de aceleración constante, no se
% usa). Los otros tres esquemas del oscilador no van (pedido 2026-10-19).
% Obstáculos y pared (imagen) son del sistema particular; su diapositiva se sacó (pedido 2026-10-21).
\begin{frame}{Modelo: ecuación de movimiento y esquema de integración}
  \setlength{\abovedisplayskip}{3pt}\setlength{\belowdisplayskip}{3pt}
  \textbf{Segunda ley de Newton con contactos elásticos}
  \[
    m_i\,\vect{a}_i(t) = \vect{F}_i(t) = \sum_{j\,:\,\xi_{ij} > 0} -k\,\xi_{ij}\,\hat{\vect{e}}_{ij},
    \qquad
    \xi_{ij} = 2r - |\vect{r}_j - \vect{r}_i|, \quad
    \hat{\vect{e}}_{ij} = \frac{\vect{r}_j - \vect{r}_i}{|\vect{r}_j - \vect{r}_i|}
  \]
  \vspace{0.3em}
  \textbf{Avance en un paso $dt$: Velocity Verlet}
  \begin{align*}
    \vect{r}_i(t + dt) &= \vect{r}_i(t) + \vect{v}_i(t)\,dt + \tfrac{1}{2}\,\vect{a}_i(t)\,dt^2\\
    \vect{v}_i(t + dt) &= \vect{v}_i(t) + \tfrac{1}{2}\bigl[\vect{a}_i(t) + \vect{a}_i(t + dt)\bigr]\,dt
  \end{align*}
\end{frame}

\begin{frame}{Observables: energía y tiempo de ejecución}

      \[
        E(t) = \sum_{i=1}^{N} \tfrac{1}{2}\,m\,|\vect{v}_i(t)|^2

      \]
    \item Error relativo de la energía, calculado sobre 10 realizaciones:
      \[
        \varepsilon = \frac{1}{K}\sum_{\ell=1}^{K} \left|\frac{E(t_\ell) - E(0)}{E(0)}\right|
      \]
    \item Tiempo de ejecución, calculado sobre 10 realizaciones
  \end{itemize}

        f(v_b, t) = \frac{N_b(t)}{10\,N\,\Delta v}, \qquad

\begin{frame}{Conclusiones}
  \begin{itemize}\setlength{\itemsep}{1em}
    \item La simulación por eventos tiene un costo inicial más bajo que por paso temporal, pero
      escala mucho más que esta con el número de partículas.

      del sistema.
