\begin{frame}{Sistema real}
  \begin{columns}[c]
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item Mesa rectangular con un arco en cada extremo y obstáculos fijos.
        \item Pelotas que se mueven libremente y chocan elásticamente entre sí,

        \item Una pelota ``convierte'' la primera vez que toca un arco; después

\begin{frame}{Modelo: esferas duras}
  Entre eventos, movimiento rectilíneo uniforme:
  $\;\vect{r}_i(t) = \vect{r}_i(t_0) + \vect{v}_i\,(t - t_0)$
  \vspace{0.2em}
  \small
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.5\textwidth}
      {\color{structure}\large Tiempo al próximo choque}\par\vspace{0.5em}
      Pared en $x_p$:
      \[
        t = \frac{x_p \mp r_i - x_i}{v_{x,i}}
      \]
      Partícula $j$ ($\sigma = r_i + r_j$, $\Delta\vect{r} = \vect{r}_j - \vect{r}_i$):
      \begin{align*}
        t_c &= -\frac{\Delta\vect{v}\cdot\Delta\vect{r} + \sqrt{d}}{\Delta\vect{v}\cdot\Delta\vect{v}}\\
        d   &= (\Delta\vect{v}\cdot\Delta\vect{r})^2 - (\Delta\vect{v}\cdot\Delta\vect{v})(\Delta\vect{r}\cdot\Delta\vect{r} - \sigma^2)
      \end{align*}
      {\small Sin choque si $\Delta\vect{v}\cdot\Delta\vect{r} \ge 0$ o $d < 0$.\\
      Obstáculo fijo: $\vect{v}_j = \vect{0}$, $\sigma = r_i + R_k$.}
    \end{column}
    \begin{column}{0.48\textwidth}
      {\color{structure}\large Después del choque (elástico)}\par\vspace{0.5em}
      Pared: se invierte la componente normal.\par\vspace{0.3em}
      Partícula $j$:
      \begin{align*}
        J &= \frac{2\,m_i m_j\,(\Delta\vect{v}\cdot\Delta\vect{r})}{\sigma\,(m_i + m_j)}\\
        \vect{v}_i' &= \vect{v}_i + \frac{J}{m_i}\,\frac{\Delta\vect{r}}{\sigma},
        \qquad
        \vect{v}_j' = \vect{v}_j - \frac{J}{m_j}\,\frac{\Delta\vect{r}}{\sigma}
      \end{align*}
      Obstáculo ($m_j \to \infty$), reflexión especular:
      \[
        \vect{v}_i' = \vect{v}_i - 2\,(\vect{v}_i\cdot\hat{\vect{e}}_n)\,\hat{\vect{e}}_n
      \]
    \end{column}
  \end{columns}
\end{frame}

\section{Implementación}
% Solo el motor. Sin post-proceso ni I/O.
% ==========================================================================

\begin{frame}[t]{Motor: cola de eventos}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.56\textwidth}
      \small
      \begin{enumerate}\setlength{\itemsep}{1pt}
        \item Para cada partícula, predecir el próximo choque contra paredes, obstáculos y
          las demás partículas; encolar cada evento con su instante $t$.
        \item Sacar el evento de menor $t$. Si alguna de sus partículas chocó después de
          predecirlo, descartarlo (\emph{contador de colisiones}).
        \item Avanzar todas las partículas hasta $t$ (MRU).
        \item Aplicar el choque. Pared corta con $|y - W/2| \le d/2$ y partícula fresca:
          gol, pasa a usada.
        \item Re-predecir solo los eventos de las partículas involucradas; volver a 2.
      \end{enumerate}
    \end{column}
    \begin{column}{0.42\textwidth}
      \centering
      \begin{tikzpicture}[>=Stealth, font=\footnotesize, node distance=0.22cm,
          caja/.style={draw, rounded corners, align=center, inner sep=2.5pt, minimum width=3.2cm}]
        \node[caja] (pq) {Cola de eventos\\ $(t,\ i,\ j,\ n_i,\ n_j)$};
        \node[caja, below=of pq] (pop) {Mínimo $t$\\ ¿$n_i$, $n_j$ vigentes?};
        \node[caja, below=of pop] (adv) {Avanzar todo a $t$};
        \node[caja, below=of adv] (col) {Choque + gol\\ $n_i{+}{+}$, $n_j{+}{+}$};
        \node[caja, below=of col] (pred) {Predecir $i$, $j$};
        \draw[->] (pq) -- (pop);
        \draw[->] (pop) -- node[right]{sí} (adv);
        \draw[->] (adv) -- (col);
        \draw[->] (col) -- (pred);
        \draw[->] (pred.east) -- ++(0.5,0) |- (pq.east);
        \draw[->] (pop.west) -- ++(-0.4,0) node[left]{no} |- (pq.west);
      \end{tikzpicture}\\[0.6em]
      {\small Fin: $t \ge t_f$ o (opcional) $F_g \ge 0.9$.}\\[0.4em]
      {\scriptsize\itshape Cola de prioridad binaria, invalidación perezosa:\\
      Sedgewick y Wayne, \emph{Algorithms}, 4.\textsuperscript{a} ed. (2011)}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Observables}
  \begin{itemize}
    \item \textbf{Tiempo de ejecución}: tiempo de reloj del lazo de eventos.
    \item \textbf{Goles}: $N_g(t)$ acumulados; $F_g(t) = N_g(t)/N$.
    \item \textbf{Tiempo al 90\,\%}: $t_{90} = \min\{t : F_g(t) \ge 0.9\}$;
      $\langle t_{90} \rangle \pm$ desvío estándar entre realizaciones.
  \end{itemize}
  \vspace{0.6em}
  \textbf{Parámetros de simulación}
  \begin{itemize}
    \item 1.1: $t_f = 30\uni{s}$, 20 realizaciones por $N$, una máquina virtual de Java
      caliente.
    \item 1.2: $N = 100$, $t_f = 100\uni{s}$, 100 realizaciones por configuración.
    \item Una semilla distinta por realización.
  \end{itemize}
\end{frame}

\section{Conclusiones}
% Solo lo que está en las figuras de Resultados. Cuantificar.
% ==========================================================================

\begin{frame}{Conclusiones}
  \begin{itemize}
    \item El tiempo de ejecución crece como $\sim N^3$ en el rango estudiado: de
      $2\ten{-3}\uni{s}$ ($N = 25$) a $16\uni{s}$ ($N = 400$) para $30\uni{s}$ simulados.
    \item Angostar la mesa a un corredor alineado con los arcos empeora, y más cuanto más
      angosto: de $21.7 \pm 2.2\uni{s}$ (vacía) a $47 \pm 6\uni{s}$ con $h = 0.20\uni{m}$.
    \item Un disco central mejora monótonamente con el radio: $16.0 \pm 2.0\uni{s}$ con
      $R = 0.33\uni{m}$, el máximo posible.
    \item Partir la mesa con una columna no cambia nada ($21 \pm 3\uni{s}$); ensanchar el
      bloque central sí: meseta de $14.3$ a $14.8\uni{s}$ para $k = 15$ a $19$. Más ancho
      ($k \ge 21$) vuelve a subir.
    \item Una o dos filas cortas en los compartimentos bajan otro $0.5\uni{s}$ (de $14.5$ a
      $14.0\uni{s}$ promediando $k = 15$ a $17$); con tres o más empeora.
    \item Elegida: $k = 17$, $n = 1$: $\langle t_{90}\rangle = 13.7 \pm 1.7\uni{s}$, un
      37\,\% menos que la mesa vacía.
  \end{itemize}
\end{frame}
