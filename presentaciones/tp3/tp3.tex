\begin{frame}{Modelo: partículas con masa}
  Entre eventos, movimiento rectilíneo uniforme:
  $\;\vect{r}_i(t) = \vect{r}_i(t_0) + \vect{v}_i\,(t - t_0)$
  \vspace{0.2em}
  \small
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.5\textwidth}
      {\color{structure}\large Tiempo al próximo choque}\par\vspace{0.5em}
      Pared en $x_p$:
      \[
        t = \frac{x_p \mp r_i - x_i}{v_{x,i}}
      \]
      Partícula $j$ ($\sigma = r_i + r_j$, $\Delta\vect{r} = \vect{r}_j - \vect{r}_i$):
      \begin{align*}
        t_c &= -\frac{\Delta\vect{v}\cdot\Delta\vect{r} + \sqrt{d}}{\Delta\vect{v}\cdot\Delta\vect{v}}\\
        d   &= (\Delta\vect{v}\cdot\Delta\vect{r})^2 - (\Delta\vect{v}\cdot\Delta\vect{v})(\Delta\vect{r}\cdot\Delta\vect{r} - \sigma^2)
      \end{align*}
      {\small Sin choque si $\Delta\vect{v}\cdot\Delta\vect{r} \ge 0$ o $d < 0$.\\
      Obstáculo fijo: $\vect{v}_j = \vect{0}$, $\sigma = r_i + R_k$.}
    \end{column}
    \begin{column}{0.48\textwidth}
      {\color{structure}\large Después del choque (elástico)}\par\vspace{0.5em}
      Pared: se invierte la componente normal.\par\vspace{0.3em}
      Partícula $j$:
      \begin{align*}
        J &= \frac{2\,m_i m_j\,(\Delta\vect{v}\cdot\Delta\vect{r})}{\sigma\,(m_i + m_j)}\\
        \vect{v}_i' &= \vect{v}_i + \frac{J}{m_i}\,\frac{\Delta\vect{r}}{\sigma},
        \qquad
        \vect{v}_j' = \vect{v}_j - \frac{J}{m_j}\,\frac{\Delta\vect{r}}{\sigma}
      \end{align*}
      Obstáculo ($m_j \to \infty$), reflexión especular:
      \[
        \vect{v}_i' = \vect{v}_i - 2\,(\vect{v}_i\cdot\hat{\vect{e}}_n)\,\hat{\vect{e}}_n
      \]
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Observables}
  \begin{itemize}
    \item \textbf{Tiempo de ejecución}: tiempo de reloj del lazo de eventos.
    \item \textbf{Goles}: $N_g(t)$ acumulados; $F_g(t) = N_g(t)/N$.
    \item \textbf{Tiempo al 90\,\%}: $t_{90} = \min\{t : F_g(t) \ge 0.9\}$.
    \item \textbf{Desplazamiento cuadrático medio}:
      $\mathrm{DCM}(t) = \frac{1}{N}\sum_i |\vect{r}_i(t) - \vect{r}_i(0)|^2$.
    \item \textbf{Difusión}: ajuste lineal $\mathrm{DCM} = 4\,D\,t$ en $[0, t_{\mathrm{fin}}]$.
  \end{itemize}
\end{frame}

\section{Conclusiones}
% Solo lo que está en las figuras de Resultados. Cuantificar.
% ==========================================================================

\begin{frame}{Conclusiones}
  \begin{itemize}
    \item Se puede reducir el tiempo de realizar 90 goles respecto de la mesa vacía utilizando
      distintas configuraciones de obstáculos.
    \item Un espacio más comprimido hacia los arcos mejora $\langle t_{90} \rangle$, pero demasiada
      compresión no permite que las partículas que ya anotaron salgan del arco y el tiempo
      requerido vuelve a aumentar.
    \item Mientras más espacio haya para el movimiento de las partículas, mayor es el DCM y, por
      lo tanto, mayor es el coeficiente de difusión $D$.
    \item En general, a mayor coeficiente de difusión $D$, mayor $\langle t_{90} \rangle$.
  \end{itemize}
\end{frame}
